📱 3. MOBILE — SOURCE SYSTEM
3.1 Хук
export function useSourceFilter() {
  const [mode, setMode] = useState("auto");

  const [sources, setSources] = useState({
    yt: true,
    sc: true,
    bc: true,
    dz: true,
  });

  const toggleSource = (key) => {
    setMode("manual");
    setSources(prev => ({ ...prev, [key]: !prev[key] }));
  };

  const setAuto = () => setMode("auto");

  return { mode, sources, toggleSource, setAuto };
}

5.3 Fetch с fallback
async function apiFetch(path, options) {
  try {
    return await fetch(API_NODES.primary + path, options);
  } catch {
    return await fetch(API_NODES.fallback + path, options);
  }
}
