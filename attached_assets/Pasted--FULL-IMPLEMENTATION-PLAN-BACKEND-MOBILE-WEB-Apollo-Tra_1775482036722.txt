3. STREAM CACHE (ускорение yt-dlp)

Создать:

src/lib/stream-cache.ts

import { redis } from "./redis";

export async function getCachedStreamUrl(id: string) {
  return await redis.get(`stream:${id}`);
}

export async function setCachedStreamUrl(id: string, url: string) {
  await redis.set(`stream:${id}`, url, "EX", 300);
}

5. WEBSOCKET (SYNC ПЛЕЕРА)

Установить:

pnpm add ws

Создать:

src/ws.ts

import { WebSocketServer } from "ws";

export const wss = new WebSocketServer({ port: 8081 });

wss.on("connection", (ws) => {
  ws.on("message", (msg) => {
    const data = JSON.parse(msg.toString());

    wss.clients.forEach(client => {
      if (client !== ws && client.readyState === 1) {
        client.send(JSON.stringify(data));
      }
    });
  });
});

Подключить в index.ts:

import "./ws";
